--Критерии выполнения л.р. 4:
--Операции РА должны быть реализованы в точном соответствии с их определением.
--Для операции "разность" нельзя использовать EXCEPT.
--Для пересечения нельзя использовать INTERSECT.
--Унарные операции должны обращаться к одной таблице, бинарные – к двум.
--Для демонстрации односхемных операций необходимо создать дополнительное отношение, схема которого совпадает со схемой одного из отношений л.р.1, и заполнить его данными.
--Не надо использовать комбинации операций (проекцию соединения, объединение селекций и т.д.).

--1. Проекция.
--это унарная операция, служащая для выбора подмножества атрибутов из отношения R. 
--Она уменьшает арность отношения и может уменьшить мощность отношения за счёт исключения одинаковых кортежей.
select DISTINCT id_tutor, name 
from v6_tutors;

--2. Селекция
--это унарная операция, результатом которой является подмножество кортежей исходного отношения, удовлетворяющих условиям назначения одного или нескольких атрибутов. 
--Арность отношения в результате селекции не меняется.
select *
from v6_tutors
where post = 'профессор';

--3. Декартово произведение
--это бинарная операция над разносхемными отношениями, соответствующая определению декартова произведения для РМД [3]. 
--Кортежи результирующего отношения состоят из всех атрибутов исходных отношений.
SELECT *
from v6_tutors, v6_students;

--4. Объединение
--объединение включает в себя все кортежи обоих исходных отношений без повторов.
CREATE TABLE v6_groups1(
    grpno NUMERIC(2) NOT NULL CONSTRAINT pk_grp1 PRIMARY KEY,
    faculty VARCHAR(80) NOT NULL
);
INSERT INTO v6_groups1
VALUES
    (6, 'прикладная химия'),
    (7, 'прикладная математика'),
    (8, 'прикладная физика'),
    (9, 'теоретическая механика'),
    (10, 'информатика'),
    (1, 'прикладная математика');

SELECT *
FROM v6_groups
UNION
SELECT *
FROM v6_groups1;
--ORDER BY grpno;

--5. Разность
--Разность – бинарная операция над односхемными отношениями. 
--Разностью T = R – S отношений R_{A1,...,An} и S_{A1,...,An} называется отношение Т, включающеемножество кортежей отношения R, не входящих в S
SELECT *
FROM v6_groups as g1
WHERE not exists(
    SELECT 1
    FROM v6_groups1 as g2
    where g1.grpno = g2.grpno
        And g1.faculty = g2.faculty
);

--6. Пересечение
--бинарная операция над односхемными отношениями.
--Пересечением T = R ∩ S отношений R_{A1,...,An} и S_{A1,...,An} называют отношение T_{A1,... An}, 
--при этом кортеж ti входит в T, если он принадлежит обоим исходным отношениям:
SELECT *
FROM v6_groups as g1
WHERE exists(
    SELECT 1
    FROM v6_groups1 as g2
    where g1.grpno = g2.grpno
        And g1.faculty = g2.faculty
);

--7. Cоединение
--– бинарная операция над разносхемными отношениями. 
--Эта операция определяет подмножество декартова произведения двух разносхемных отношений. 
--Кортеж декартова произведения входит в результирующее отношение, 
--если выполняется условие соединения F, которое задаёт соотношение значений атрибутов разных таблиц.
SELECT s.*, g.faculty
FROM v6_students as s, v6_groups as g
WHERE g.grpno = s.grpno;
